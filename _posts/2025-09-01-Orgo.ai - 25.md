```
// Include core libigl functions for mesh processing
#include <igl/read_triangle_mesh.h>    // load .obj/.off mesh
#include <igl/per_vertex_normals.h>    // compute per-vertex normals
#include <igl/per_face_normals.h>      // compute per-face normals
#include <igl/cotmatrix.h>             // build cotangent Laplacian
#include <igl/massmatrix.h>            // build mass (area) matrix
#include <igl/gaussian_curvature.h>    // compute Gaussian curvature
#include <igl/principal_curvature.h>   // compute principal curvatures
#include <igl/jet.h>                   // color mapping function
#include <igl/viewer/Viewer.h>         // simple mesh viewer

// Eigen: linear algebra library (matrices, vectors, solvers)
#include <Eigen/Core>
#include <Eigen/Sparse>
#include <iostream>

using namespace Eigen;

int main(int argc, char *argv[])
{
    // -------------------------------
    // [Step 1] Input Mesh
    // -------------------------------

    MatrixXd V;   // (n x 3) matrix of vertex coordinates
    MatrixXi F;   // (m x 3) matrix of face indices (triangles)
    igl::read_triangle_mesh("cow.obj", V, F);  
    // Reads a mesh from file into V (vertices) and F (faces)

    std::cout << "Loaded mesh: " 
              << V.rows() << " vertices, " 
              << F.rows() << " faces\n";  
    // .rows() = number of rows in Eigen matrix

    // -------------------------------
    // [Step 2] Normals
    // -------------------------------

    MatrixXd FN; // per-face normals (m x 3)
    MatrixXd VN; // per-vertex normals (n x 3)

    igl::per_face_normals(V, F, FN);  
    // Computes one normal per face using cross product of edges

    igl::per_vertex_normals(V, F, VN);  
    // Computes vertex normals as weighted average of incident faces

    // Note: PCA-based or quadratic fitting normals 
    // require custom implementation using igl::fit_plane or igl::principal_curvature

    // -------------------------------
    // [Step 3] Laplace–Beltrami Operator
    // -------------------------------

    SparseMatrix<double> L;  
    // Sparse Laplacian matrix (n x n), most entries are zero
    igl::cotmatrix(V, F, L);  
    // Build cotangent-weighted Laplacian

    SparseMatrix<double> M;  
    // Sparse mass matrix (n x n), encodes vertex areas
    igl::massmatrix(V, F, igl::MASSMATRIX_TYPE_VORONOI, M);  
    // Build Voronoi area-based mass matrix

    // -------------------------------
    // [Step 4] Discrete Curvature
    // -------------------------------

    VectorXd K;  
    // Gaussian curvature per vertex (n x 1)
    igl::gaussian_curvature(V, F, K);  
    // Computes angle deficit at each vertex

    MatrixXd PD1, PD2; // principal curvature directions
    VectorXd PV1, PV2; // principal curvature values
    igl::principal_curvature(V, F, PD1, PD2, PV1, PV2);  
    // Computes min/max curvature and directions

    VectorXd H = 0.5 * (PV1 + PV2);  
    // Mean curvature = average of principal curvatures

    MatrixXd C;
    igl::jet(H, true, C);  
    // Maps curvature values to RGB colors (for visualization)

    // -------------------------------
    // [Step 5] Mesh Smoothing
    // -------------------------------

    // Explicit Laplacian smoothing (forward Euler)
    double lambda = 0.01;  
    // Step size (too large will cause instability)
    MatrixXd V_explicit = V;  
    // Copy original vertices
    for (int iter = 0; iter < 100; iter++) {
        MatrixXd LV = L * V_explicit;  
        // Apply Laplacian operator to vertices
        V_explicit += lambda * LV;  
        // Update vertex positions
    }

    // Implicit Laplacian smoothing (backward Euler)
    double t = 20.0;  
    // Large timestep (stable for implicit method)
    SparseMatrix<double> S = M - t * L;  
    // System matrix: (M - tL)
    SimplicialLDLT<SparseMatrix<double>> solver;  
    // Eigen sparse linear solver (Cholesky factorization)
    solver.compute(S);
    MatrixXd V_implicit = solver.solve(M * V);  
    // Solve (M - tL) V_new = M V

    // -------------------------------
    // Launch Viewer
    // -------------------------------

    igl::viewer::Viewer viewer;
    viewer.data().set_mesh(V, F);   // show original mesh
    viewer.data().set_colors(C);    // color mesh by mean curvature
    viewer.launch();                // start interactive viewer

    return 0;
}
```

```
[Step 1] Input Mesh
   - Load mesh and initialize data structures
        ↓
[Step 2] Differential Quantities
   - Compute per-face and per-vertex normals
   - Approximate curvature quantities
        ↓
[Step 3] Laplace–Beltrami Operator
   - Build cotangent Laplacian with area weights
        ↓
[Step 4] Curvature Estimation
   - Mean, Gaussian, principal curvatures
   - Visualize curvature with color maps
        ↓
[Step 5] Mesh Smoothing
   - Explicit vs Implicit Laplacian smoothing
   - Bilateral denoising
        ↓
[Applications]
   - Mesh fairing & denoising
   - Parameterization & deformation
   - Geometry-aware shape analysis
```
